import json,subprocess,concurrent.futures
from pathlib import Path
from datetime import datetime,timezone
origin='https://app-governance-validator-atiya-1026.azurewebsites.net'
def fetch(route):
 try:
  p=subprocess.run(['curl','--silent','--show-error','--fail-with-body','--connect-timeout','5','--max-time','35','-H','Cache-Control: no-cache',origin+route],capture_output=True,text=True,timeout=40)
  return route,{'curl_exit_code':p.returncode,'body':json.loads(p.stdout) if p.returncode==0 else None}
 except Exception as e:return route,{'error_kind':type(e).__name__}
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:r=dict(pool.map(fetch,['/health','/ready','/api/sso/status']))
r['observed_at_utc']=datetime.now(timezone.utc).isoformat();Path('docs/evidence/stage-f-live-worker-2026-11-05/http.json').write_text(json.dumps(r,indent=2)+'\n');print(json.dumps(r))
